# ---------------------------------------------------------------------------
# Embed runtime_patches/builtin_default.tjs into a generated C++ header
# (RUNTIME_PATCH_DEFAULT_TJS) that runtimePatchLoader.cpp consumes.
# Add new built-in compatibility patches here:
#   1. Drop the .tjs file in cpp/plugins/runtime_patches/
#   2. Append it to RUNTIME_PATCH_FILES below
#   3. Reference the generated symbol in runtimePatchLoader.cpp
# ---------------------------------------------------------------------------
set(RUNTIME_PATCH_FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/runtime_patches/builtin_default.tjs"
)

set(RUNTIME_PATCH_GENERATED_DIR
    "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime_patches")
file(MAKE_DIRECTORY "${RUNTIME_PATCH_GENERATED_DIR}")
set(RUNTIME_PATCH_HEADER
    "${RUNTIME_PATCH_GENERATED_DIR}/runtime_patch_strings.h")

set(RUNTIME_PATCH_HEADER_CONTENT
    "// Auto-generated by cpp/plugins/CMakeLists.txt — do not edit\n#pragma once\n\n")
foreach(_patch_file IN LISTS RUNTIME_PATCH_FILES)
    get_filename_component(_patch_stem "${_patch_file}" NAME_WE)
    file(READ "${_patch_file}" _patch_body)
    # Escape backslashes / quotes / newlines for a C++ raw-byte literal.
    string(REPLACE "\\" "\\\\" _patch_body "${_patch_body}")
    string(REPLACE "\"" "\\\"" _patch_body "${_patch_body}")
    string(REPLACE "\n" "\\n\"\n    \"" _patch_body "${_patch_body}")
    string(TOUPPER "${_patch_stem}" _patch_macro)
    string(APPEND RUNTIME_PATCH_HEADER_CONTENT
        "// Embedded from ${_patch_file}\nstatic const char *const RUNTIME_PATCH_${_patch_macro}_TJS =\n    \"${_patch_body}\";\n\n")
endforeach()
file(WRITE "${RUNTIME_PATCH_HEADER}" "${RUNTIME_PATCH_HEADER_CONTENT}")

target_include_directories(${PROJECT_NAME} PRIVATE
    "${RUNTIME_PATCH_GENERATED_DIR}"
)
# Re-run CMake configuration if any embedded .tjs changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${RUNTIME_PATCH_FILES})

option(KRKR2_ENABLE_LIVE2D "Build the optional Live2D Cubism plugin when its SDK is complete" ON)
if(KRKR2_ENABLE_LIVE2D)
    set(KRKR2_LIVE2D_DIR "${CMAKE_CURRENT_SOURCE_DIR}/live2d")
    set(KRKR2_LIVE2D_CUBISM_DIR "${KRKR2_LIVE2D_DIR}/cubism")
    set(KRKR2_LIVE2D_CORE_DIR "" CACHE PATH "Path to a local Live2D Cubism Core SDK directory")
    if(NOT KRKR2_LIVE2D_CORE_DIR AND DEFINED ENV{KRKR2_LIVE2D_CORE_DIR})
        set(KRKR2_LIVE2D_CORE_DIR "$ENV{KRKR2_LIVE2D_CORE_DIR}")
    endif()
    if(NOT KRKR2_LIVE2D_CORE_DIR)
        if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/live2d-sdk/Core")
            set(KRKR2_LIVE2D_CORE_DIR "${CMAKE_SOURCE_DIR}/third_party/live2d-sdk/Core")
        else()
            set(KRKR2_LIVE2D_CORE_DIR "${KRKR2_LIVE2D_CUBISM_DIR}/Core")
        endif()
    endif()
    set(KRKR2_LIVE2D_FRAMEWORK_DIR "${KRKR2_LIVE2D_CUBISM_DIR}/Framework")
    set(KRKR2_LIVE2D_MISSING)

    foreach(_live2d_required_file IN ITEMS
            "${KRKR2_LIVE2D_DIR}/krkrlive2d.cpp"
            "${KRKR2_LIVE2D_DIR}/krkrgles.cpp"
            "${KRKR2_LIVE2D_FRAMEWORK_DIR}/CubismFramework.hpp"
            "${KRKR2_LIVE2D_FRAMEWORK_DIR}/CubismModelSettingJson.hpp"
            "${KRKR2_LIVE2D_CORE_DIR}/include/Live2DCubismCore.h")
        if(NOT EXISTS "${_live2d_required_file}")
            list(APPEND KRKR2_LIVE2D_MISSING "${_live2d_required_file}")
        endif()
    endforeach()

    if(ANDROID)
        set(KRKR2_LIVE2D_ANDROID_ABI "${CMAKE_ANDROID_ARCH_ABI}")
        if(NOT KRKR2_LIVE2D_ANDROID_ABI)
            set(KRKR2_LIVE2D_ANDROID_ABI "${ANDROID_ABI}")
        endif()
        set(KRKR2_LIVE2D_CORE_LIB
            "${KRKR2_LIVE2D_CORE_DIR}/lib/android/${KRKR2_LIVE2D_ANDROID_ABI}/libLive2DCubismCore.a")
        if(NOT EXISTS "${KRKR2_LIVE2D_CORE_LIB}")
            list(APPEND KRKR2_LIVE2D_MISSING "${KRKR2_LIVE2D_CORE_LIB}")
        endif()
    else()
        list(APPEND KRKR2_LIVE2D_MISSING "Android target")
    endif()

    if(KRKR2_LIVE2D_MISSING)
        list(JOIN KRKR2_LIVE2D_MISSING ", " KRKR2_LIVE2D_MISSING_TEXT)
        message(STATUS "Live2D Cubism plugin disabled; missing: ${KRKR2_LIVE2D_MISSING_TEXT}")
    else()
        find_package(PNG REQUIRED)
        find_package(LibArchive REQUIRED)

        file(GLOB_RECURSE KRKR2_LIVE2D_FRAMEWORK_SOURCES CONFIGURE_DEPENDS
            "${KRKR2_LIVE2D_FRAMEWORK_DIR}/*.cpp")
        list(FILTER KRKR2_LIVE2D_FRAMEWORK_SOURCES EXCLUDE REGEX
            ".*/D3D9/.*|.*/D3D11/.*|.*/Vulkan/.*|.*/Metal/.*|.*/Cocos2d/.*")

        add_library(krkr2_cubism_framework STATIC ${KRKR2_LIVE2D_FRAMEWORK_SOURCES})
        target_include_directories(krkr2_cubism_framework PUBLIC
            "${KRKR2_LIVE2D_FRAMEWORK_DIR}"
            "${KRKR2_LIVE2D_CORE_DIR}/include")
        target_compile_definitions(krkr2_cubism_framework PUBLIC
            CSM_TARGET_ANDROID_ES2)
        target_link_libraries(krkr2_cubism_framework PUBLIC
            "${KRKR2_LIVE2D_CORE_LIB}"
            EGL
            GLESv3)

        target_sources(${PROJECT_NAME} PRIVATE
            live2d/krkrlive2d.cpp
            live2d/krkrgles.cpp)
        target_include_directories(${PROJECT_NAME} PRIVATE
            "${KRKR2_LIVE2D_DIR}"
            "${KRKR2_LIVE2D_FRAMEWORK_DIR}"
            "${KRKR2_LIVE2D_CORE_DIR}/include")
        target_compile_definitions(${PROJECT_NAME} PUBLIC
            CSM_TARGET_ANDROID_ES2)
        target_link_libraries(${PROJECT_NAME} PUBLIC
            krkr2_cubism_framework
            LibArchive::LibArchive
            PNG::PNG
            EGL
            GLESv3)
    endif()
endif()
