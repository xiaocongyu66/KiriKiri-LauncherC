# ---------------------------------------------------------------------------
# Embed runtime_patches/builtin_default.tjs into a generated C++ header
# (RUNTIME_PATCH_DEFAULT_TJS) that runtimePatchLoader.cpp consumes.
# Add new built-in compatibility patches here:
#   1. Drop the .tjs file in cpp/plugins/runtime_patches/
#   2. Append it to RUNTIME_PATCH_FILES below
#   3. Reference the generated symbol in runtimePatchLoader.cpp
# ---------------------------------------------------------------------------
set(RUNTIME_PATCH_FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/runtime_patches/builtin_default.tjs"
)

set(RUNTIME_PATCH_GENERATED_DIR
    "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime_patches")
file(MAKE_DIRECTORY "${RUNTIME_PATCH_GENERATED_DIR}")
set(RUNTIME_PATCH_HEADER
    "${RUNTIME_PATCH_GENERATED_DIR}/runtime_patch_strings.h")

set(RUNTIME_PATCH_HEADER_CONTENT
    "// Auto-generated by cpp/plugins/CMakeLists.txt — do not edit\n#pragma once\n\n")
foreach(_patch_file IN LISTS RUNTIME_PATCH_FILES)
    get_filename_component(_patch_stem "${_patch_file}" NAME_WE)
    file(READ "${_patch_file}" _patch_body)
    # Escape backslashes / quotes / newlines for a C++ raw-byte literal.
    string(REPLACE "\\" "\\\\" _patch_body "${_patch_body}")
    string(REPLACE "\"" "\\\"" _patch_body "${_patch_body}")
    string(REPLACE "\n" "\\n\"\n    \"" _patch_body "${_patch_body}")
    string(TOUPPER "${_patch_stem}" _patch_macro)
    string(APPEND RUNTIME_PATCH_HEADER_CONTENT
        "// Embedded from ${_patch_file}\nstatic const char *const RUNTIME_PATCH_${_patch_macro}_TJS =\n    \"${_patch_body}\";\n\n")
endforeach()
file(WRITE "${RUNTIME_PATCH_HEADER}" "${RUNTIME_PATCH_HEADER_CONTENT}")

target_include_directories(${PROJECT_NAME} PRIVATE
    "${RUNTIME_PATCH_GENERATED_DIR}"
)
# Re-run CMake configuration if any embedded .tjs changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${RUNTIME_PATCH_FILES})
